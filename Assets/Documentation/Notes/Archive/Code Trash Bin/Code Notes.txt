{
    float Repeat(double t, double length) => (float) Math.Clamp(t - Math.Floor(t / length) * length, 0.0, length) / length;
    fixedUpdateLapse = Repeat(Time.timeAsDouble, Time.fixedDeltaTime);

private void UpdateFollowCamera(CameraView view, bool enableSideOffset)
{
    // Side offset
    targetSideOffset = -shipPhysics.SideSpeedMPS;
    currentSideOffset = Mathf.SmoothDamp(currentSideOffset, targetSideOffset, ref sideOffsetVelocity, sideOffsetSmooth);
    sideOffset = enableSideOffset? new Vector3(sideSteerOffset * currentSideOffset, 0, 0) : Vector3.zero;
    
    
    
    // Calculate camera offset, pivot and rotation
    offsetToPivot = view.target.rotation * Vector3.forward * view.pivotDistance;
    pivotTransform.position = view.target.TransformPoint(view.offset + sideOffset + extraOffset) + offsetToPivot;
    pivotTransform.rotation = view.target.rotation * Quaternion.Euler(extraPivotRotation);
    
    // Apply camera offset, pivot and rotation
    targetCamPosition = pivotTransform.TransformPoint(Vector3.forward * -view.pivotDistance);
    targetCamRotation = pivotTransform.rotation * Quaternion.Euler(view.rotation + extraRotation);
    cameraTransform.SetPositionAndRotation(targetCamPosition, targetCamRotation);
}
